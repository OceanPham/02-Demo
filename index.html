// Real-time Booking & Department Sync Hook
export function useRealtimeReservation(slotId: string) {
  useEffect(() => {
    socket.emit("slot:Hold", { slotId, ttlSeconds: 300 });
    socket.on("slot:StatusChanged", (payload) => {
      queryClient.setQueryData(["slot", slotId], payload);
    });
    return () => socket.off("slot:StatusChanged");
  }, [slotId]);
}
